##### ------ PREPARE GENOME DATABASE ------ ##### 

### Keep only genomes listed in "inc.txt".

sed -e 's/$/.fasta/' -i inc.txt
xargs -a inc.txt mv -t paper

### Reformat genome names and save to one file.

awk '/^>/{print ">" substr(FILENAME,1,length(FILENAME)-6) "_" ++i ; next} 1' *.fasta > bc_contigs.fasta


### Prepare genomes as BLAST database.

makeblastdb -in bc_contigs.fasta -dbtype nucl -parse_seqids 


##### ------ EXTRACT UPSTREAM SEQUENCES ------ #####


### Run BLAST for the first genes in the nhe and hbl operons (nheA and hblC).
### nheA from NZ_JBJJSR010000001.1:1765215-1766375
### hblC from NZ_JBJJSR010000001.1:3066802-3065483

blastn -db bc_contigs.fasta -query nheA_query.fasta -max_target_seqs 10000 -perc_identity 60 -qcov_hsp_perc 70 -task megablast -outfmt 7 -out nheA_blast.csv 
blastn -db bc_contigs.fasta -query hblC_query.fasta -max_target_seqs 10000 -perc_identity 60 -qcov_hsp_perc 70 -task megablast -outfmt 7 -out hblC_blast.csv 
blastn -db bc_contigs.fasta -query cytK2_query.fasta -max_target_seqs 10000 -perc_identity 60 -qcov_hsp_perc 70 -task megablast -outfmt 7 -out cytK2_blast.csv
blastn -db bc_contigs.fasta -query cytK1_query.fasta -max_target_seqs 10000 -perc_identity 60 -qcov_hsp_perc 70 -task megablast -outfmt 7 -out cytK1_blast.csv

### Separate genomes with enterotoxin genes on the + and - strands for sequence extraction.

awk -F'\t' -v OFS='\t' '$9 - $10 > 0' nheA_blast.csv > nheA_minus.csv
awk -F'\t' -v OFS='\t' '$9 - $10 < 0' nheA_blast.csv > nheA_plus.csv

awk -F'\t' -v OFS='\t' '$9 - $10 > 0' hblC_blast.csv > hblC_minus.csv
awk -F'\t' -v OFS='\t' '$9 - $10 < 0' hblC_blast.csv > hblC_plus.csv

awk -F'\t' -v OFS='\t' '$9 - $10 > 0' cytK2_blast.csv > cytK2_minus.csv
awk -F'\t' -v OFS='\t' '$9 - $10 < 0' cytK2_blast.csv > cytK2_plus.csv

awk -F'\t' -v OFS='\t' '$9 - $10 > 0' cytK1_blast.csv > cytK1_minus.csv
awk -F'\t' -v OFS='\t' '$9 - $10 < 0' cytK1_blast.csv > cytK1_plus.csv

### Calculate 600, 1500, or 150 bp upstream of the BLAST start location and save to column 9. If the value is negative because the end of a contig is reached, set number to "1".

awk -F'\t' -v OFS='\t' '{new_val=($9-600) < 1 ? 1 :($9-600); print $0, new_val}' nheA_plus.csv > nheA_plus_2.csv
awk -F'\t' -v OFS='\t' '{$3 = $9 "\t" $3; $3=($3+600); print }' nheA_minus.csv > nheA_minus_2.csv

awk -F'\t' -v OFS='\t' '{new_val=($9-1500) < 1 ? 1 :($9-1500); print $0, new_val}' hblC_plus.csv > hblC_plus_2.csv
awk -F'\t' -v OFS='\t' '{$3 = $9 "\t" $3; $3=($3+1500); print }' hblC_minus.csv > hblC_minus_2.csv

awk -F'\t' -v OFS='\t' '{new_val=($9-150) < 1 ? 1 :($9-150); print $0, new_val}' cytK2_plus.csv > cytK2_up_plus_2.csv
awk -F'\t' -v OFS='\t' '{$3 = $9 "\t" $3; $3=($3+150); print }' cytK2_minus.csv > cytK2_up_minus_2.csv

awk -F'\t' -v OFS='\t' '{new_val=($9-150) < 1 ? 1 :($9-150); print $0, new_val}' cytK1_plus.csv > cytK1_up_plus_2.csv
awk -F'\t' -v OFS='\t' '{$3 = $9 "\t" $3; $3=($3+150); print }' cytK1_minus.csv > cytK1_up_minus_2.csv


### Extract sequences 600, 1500, or 150 bp upstream of nheA, hblC, cytK respectively from BLAST start location.

awk '{OFS="\t"}{print $2,$9,$13}' nheA_plus_2.csv | xargs -n 3 sh -c 'blastdbcmd -db /workdir/crp86/paper/bc_contigs.fasta -entry "$0" -range "$2"-"$1" -outfmt %f >> nheA_up_plus.fasta' 
awk '{OFS="\t"}{print $2,$3,$9}' nheA_minus_2.csv | xargs -n 3 sh -c 'blastdbcmd -db /workdir/crp86/paper/bc_contigs.fasta -entry "$0" -range "$2"-"$1" -outfmt %f >> nheA_up_minus.fasta' 

awk '{OFS="\t"}{print $2,$9,$13}' hblC_plus_2.csv | xargs -n 3 sh -c 'blastdbcmd -db /workdir/crp86/paper/bc_contigs.fasta -entry "$0" -range "$2"-"$1" -outfmt %f >> hblC_up_plus.fasta' 
awk '{OFS="\t"}{print $2,$3,$9}' hblC_minus_2.csv | xargs -n 3 sh -c 'blastdbcmd -db /workdir/crp86/paper/bc_contigs.fasta -entry "$0" -range "$2"-"$1" -outfmt %f >> hblC_up_minus.fasta' 

awk '{OFS="\t"}{print $2,$9,$13}' cytK2_up_plus_2.csv | xargs -n 3 sh -c 'blastdbcmd -db /workdir/crp86/paper/bc_contigs.fasta -entry "$0" -range "$2"-"$1" -outfmt %f >> cytK2_up_plus.fasta' 
awk '{OFS="\t"}{print $2,$3,$9}' cytK2_up_minus_2.csv | xargs -n 3 sh -c 'blastdbcmd -db /workdir/crp86/paper/bc_contigs.fasta -entry "$0" -range "$2"-"$1" -outfmt %f >> cytK2_up_minus.fasta' 

awk '{OFS="\t"}{print $2,$9,$13}' cytK1_up_plus_2.csv | xargs -n 3 sh -c 'blastdbcmd -db /workdir/crp86/paper/bc_contigs.fasta -entry "$0" -range "$2"-"$1" -outfmt %f >> cytK1_up_plus.fasta' 
awk '{OFS="\t"}{print $2,$3,$9}' cytK1_up_minus_2.csv | xargs -n 3 sh -c 'blastdbcmd -db /workdir/crp86/paper/bc_contigs.fasta -entry "$0" -range "$2"-"$1" -outfmt %f >> cytK1_up_minus.fasta' 


### Reverse complement sequences from - strands.

export PATH=/programs/seqtk:$PATH

seqtk seq -r /workdir/crp86/paper/blast_3_3_26/nheA_minus.fasta > nheA_minus_R.fasta 
seqtk seq -r /workdir/crp86/paper/blast_3_3_26/hblC_minus.fasta > hblC_minus_R.fasta 
seqtk seq -r /workdir/crp86/paper/blast_3_3_26/cytK2_up_minus.fasta > cytK2_up_minus_R.fasta 
seqtk seq -r /workdir/crp86/paper/blast_3_3_26/cytK1_up_minus.fasta > cytK1_up_minus_R.fasta 

### Save the + and - sequences to one file.

cat nheA_up_plus.fasta nheA_up_minus_R.fasta > nheA_up_all.fasta

#!/bin/bash

export PATH=/programs/seqtk:$PATH

for file in *_query.fasta; do
        prefix="${file%%_*}"
        echo $prefix
        blastn -db /workdir/crp86/paper/bc_contigs.fasta -query $file -max_target_seqs 10000 -perc_identity 60 -qcov_hsp_perc 70 -task megablast -outfmt 7 -out ${prefix}_blast.csv
        echo ${prefix}_blast.csv
        head ${prefix}_blast.csv
	
	awk -F'\t' -v OFS='\t' '$9 - $10 < 0' ${prefix}_blast.csv > ${prefix}_blast_plus.csv
	awk -F'\t' -v OFS='\t' '$9 - $10 > 0' ${prefix}_blast.csv > ${prefix}_blast_minus.csv

	awk '{OFS="\t"}{print $2,$9,$10}' ${prefix}_blast_plus.csv | xargs -n 3 sh -c 'blastdbcmd -db /workdir/crp86/paper/bc_contigs.fasta -entry "$0" -range "$1"-"$2" -outfmt %f >> "'"${prefix}_plus.fasta"'"'
	awk '{OFS="\t"}{print $2,$9,$10}' ${prefix}_blast_minus.csv | xargs -n 3 sh -c 'blastdbcmd -db /workdir/crp86/paper/bc_contigs.fasta -entry "$0" -range "$2"-"$1" -outfmt %f >> "'"${prefix}_minus.fasta"'"'
	seqtk seq -r /workdir/crp86/paper/blast_3_3_26/${prefix}_minus.fasta > ${prefix}_minus_R.fasta 
	cat ${prefix}_plus.fasta  ${prefix}_minus_R.fasta  > ${prefix}_all.fasta 
	head ${prefix}_all.fasta
